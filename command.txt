CREATE: coupon.json

'{"coupon_id":"coupon-003","name":"Presigned URL Test Coupon","discount":25}' | Set-Content -Path .\coupon.json -Encoding ascii

$response = Invoke-RestMethod `
    -Uri "http://localhost:4567/restapis/jt04umn17f/dev/_user_request_/coupons/import_presigned_url" `
    -Method POST `
    -ContentType "application/json" `
    -Body '{"file_name":"coupon.json"}'

$uploadUrl = $response.url.Replace(":4566/", ":4567/")

Invoke-RestMethod `
    -Uri $uploadUrl `
    -Method PUT `
    -ContentType "application/json" `
    -InFile ".\coupon.json"

aws s3 ls s3://coupons/ `
    --endpoint-url=http://localhost:4567 `
    --region us-east-1
